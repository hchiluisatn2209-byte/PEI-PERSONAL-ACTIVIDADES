-- ============================================================ -- Fix: RLS para rrhh.disponibilidades -- Ejecutar en Supabase → SQL Editor -- ============================================================ -- Ver políticas actuales (para saber qué existe) SELECT policyname, cmd, qual, with_check FROM pg_policies WHERE schemaname = 'rrhh' AND tablename = 'disponibilidades'; -- Opción A (recomendada para empezar): permitir todo a usuarios autenticados CREATE POLICY "allow_insert_authenticated" ON rrhh.disponibilidades FOR INSERT TO authenticated WITH CHECK (true); CREATE POLICY "allow_update_authenticated" ON rrhh.disponibilidades FOR UPDATE TO authenticated USING (true) WITH CHECK (true); CREATE POLICY "allow_select_authenticated" ON rrhh.disponibilidades FOR SELECT TO authenticated USING (true); -- Si ya existen políticas con esos nombres, elimínalas primero: -- DROP POLICY IF EXISTS "allow_insert_authenticated" ON rrhh.disponibilidades; -- DROP POLICY IF EXISTS "allow_update_authenticated" ON rrhh.disponibilidades; -- DROP POLICY IF EXISTS "allow_select_authenticated" ON rrhh.disponibilidades;
